-- Mugerwa Hope Foundation: cloud posts, post likes/comments and subscribers -- Run this ONCE in Supabase SQL Editor. -- It is designed for the existing Supabase project used by the website. create extension if not exists pgcrypto; create table if not exists public.mhf_posts ( id uuid primary key default gen_random_uuid(), title text not null, body text not null, post_date date not null default current_date, published boolean not null default true, author_id uuid references auth.users(id) on delete set null, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); create index if not exists mhf_posts_public_date_idx on public.mhf_posts (published, post_date desc, created_at desc); create table if not exists public.mhf_post_likes ( id uuid primary key default gen_random_uuid(), post_id uuid not null references public.mhf_posts(id) on delete cascade, visitor_token text not null, created_at timestamptz not null default now(), unique(post_id, visitor_token) ); create table if not exists public.mhf_post_comments ( id uuid primary key default gen_random_uuid(), post_id uuid not null references public.mhf_posts(id) on delete cascade, visitor_name text not null, visitor_email text, body text not null, status text not null default 'pending' check (status in ('pending','approved','rejected')), approved_at timestamptz, approved_by uuid references auth.users(id) on delete set null, created_at timestamptz not null default now() ); create index if not exists mhf_post_comments_post_idx on public.mhf_post_comments (post_id, created_at desc); create table if not exists public.mhf_subscribers ( id uuid primary key default gen_random_uuid(), email text not null, subscribed_at timestamptz not null default now(), unique(email) ); alter table public.mhf_posts enable row level security; alter table public.mhf_post_likes enable row level security; alter table public.mhf_post_comments enable row level security; alter table public.mhf_subscribers enable row level security; -- Public visitors may see published posts. drop policy if exists "Public can read published MHF posts" on public.mhf_posts; create policy "Public can read published MHF posts" on public.mhf_posts for select to anon, authenticated using (published = true or (auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com'); -- Only the foundation administrator may create/change/delete posts. drop policy if exists "Admin manages MHF posts" on public.mhf_posts; create policy "Admin manages MHF posts" on public.mhf_posts for all to authenticated using ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com') with check ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com'); -- Public likes: anyone can add a like and see like rows/counts. drop policy if exists "Public can read MHF post likes" on public.mhf_post_likes; create policy "Public can read MHF post likes" on public.mhf_post_likes for select to anon, authenticated using (true); drop policy if exists "Public can add MHF post likes" on public.mhf_post_likes; create policy "Public can add MHF post likes" on public.mhf_post_likes for insert to anon, authenticated with check (visitor_token is not null and length(visitor_token) > 5); -- Only admin can delete likes if needed. drop policy if exists "Admin can delete MHF post likes" on public.mhf_post_likes; create policy "Admin can delete MHF post likes" on public.mhf_post_likes for delete to authenticated using ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com'); -- Visitors can submit comments; public can only read approved comments. drop policy if exists "Public can read approved MHF post comments" on public.mhf_post_comments; create policy "Public can read approved MHF post comments" on public.mhf_post_comments for select to anon, authenticated using ( status = 'approved' or (auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com' ); drop policy if exists "Public can submit MHF post comments" on public.mhf_post_comments; create policy "Public can submit MHF post comments" on public.mhf_post_comments for insert to anon, authenticated with check (status = 'pending'); drop policy if exists "Admin manages MHF post comments" on public.mhf_post_comments; create policy "Admin manages MHF post comments" on public.mhf_post_comments for update to authenticated using ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com') with check ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com'); drop policy if exists "Admin deletes MHF post comments" on public.mhf_post_comments; create policy "Admin deletes MHF post comments" on public.mhf_post_comments for delete to authenticated using ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com'); -- Visitors can subscribe. Only admin can read/delete subscribers. drop policy if exists "Public can subscribe to MHF updates" on public.mhf_subscribers; create policy "Public can subscribe to MHF updates" on public.mhf_subscribers for insert to anon, authenticated with check (email is not null); drop policy if exists "Admin can read MHF subscribers" on public.mhf_subscribers; create policy "Admin can read MHF subscribers" on public.mhf_subscribers for select to authenticated using ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com'); drop policy if exists "Admin can delete MHF subscribers" on public.mhf_subscribers; create policy "Admin can delete MHF subscribers" on public.mhf_subscribers for delete to authenticated using ((auth.jwt()->>'email') = 'mugerwalouiscupa@gmail.com'); -- Publish a simple test post after the tables are created if you want: -- insert into public.mhf_posts(title, body, post_date, published) -- values ('Welcome to Mugerwa Hope Foundation', -- 'This is our first public foundation update.', -- current_date, true);
